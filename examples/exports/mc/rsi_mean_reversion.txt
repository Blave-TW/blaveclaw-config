Inputs:
    Len(14),
    BuyLevel(30),          // Blave BUY_TH
    ExitLevel(55),         // Blave EXIT_TH  (must be > BuyLevel)
    Qty(1);

Variables:
    RSIval(0);

// --- indicators ---
RSIval = RSI(Close, Len);

// --- signal ---
Condition1 = RSIval < BuyLevel;     // Blave: signal = 1.0
Condition2 = RSIval > ExitLevel;    // Blave: signal = 0.0

// --- orders ---
