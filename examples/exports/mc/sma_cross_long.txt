Inputs:
    FastLen(45),          // Blave SMA_FAST
    SlowLen(100),         // Blave SMA_SLOW
    Qty(1);               // fixed size; Blave sizes as a fraction of equity

Variables:
    FastMA(0),
    SlowMA(0);

// --- indicators ---
// Call series functions unconditionally, every bar, before any branch.
FastMA = Average(Close, FastLen);
SlowMA = Average(Close, SlowLen);

// --- signal ---
Condition1 = FastMA > SlowMA;      // Blave: signal = 1.0
Condition2 = FastMA < SlowMA;      // Blave: signal = 0.0
// FastMA = SlowMA → neither: hold (Blave nan)

// --- orders ---
if Condition1 and MarketPosition <= 0 then
    Buy ("LE") Qty contracts next bar at market;

if Condition2 and MarketPosition = 1 then
    Sell ("LX") next bar at market;
